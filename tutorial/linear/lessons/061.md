## 什么时候会用到这一页

外部系统通过HTTP把review任务发送给已有会话。先启动本机Webhook服务，再发送签名请求，观察任务落入Journal并由真实Agent调用离线模型；重复投递只接收一次。这是自定义协议，不是GitHub官方Webhook格式。

### 这次数据怎么走

```text
签名发送脚本 → POST本机/webhooks → HTTP层读取原始字节和请求头 → webhook验签/去重 → Journal保存inbox/add → Agent.pump领取任务 → runTurn调用OfflineModel → Session记录回复与回合结束。HTTP回执只表示接收状态。
```

### 创建 [`src/webhook-http.mjs`](../../../harness-own/src/webhook-http.mjs)

````javascript
import { createServer } from 'node:http'
import { webhook } from './webhook.mjs'

/** 启动独立的本机Webhook入口；固定绑定一个已有会话，不接受请求指定owner或工作区。 */
export async function startWebhook(record, { secret, port = 0, maxBytes = 64 * 1024 } = {}) {
  const receive = webhook({ secret })
  // 响应只报告接收结果，模型回复从Session事件读取，不在HTTP回执中假装完成。
  const server = createServer(async (request, response) => {
    const reply = (status, value) => {
      response.writeHead(status, { 'content-type': 'application/json; charset=utf-8' })
      response.end(JSON.stringify(value))
    }
    try {
      if (request.url !== '/webhooks') { request.resume(); reply(404, { error: '未找到路由' }); return }
      if (request.method !== 'POST') { request.resume(); response.setHeader('allow', 'POST'); reply(405, { error: '只接受POST' }); return }
      if (request.headers['content-type']?.split(';')[0].trim().toLowerCase() !== 'application/json') {
        request.resume(); reply(415, { error: '需要application/json' }); return
      }
      // 保留原始字节验签，并限制正文大小；不能先JSON.parse再序列化后验证。
      const chunks = []; let size = 0
      for await (const chunk of request) {
        size += chunk.length
        if (size > maxBytes) { request.resume(); reply(413, { error: 'Webhook正文过大' }); return }
        chunks.push(chunk)
      }
      const result = receive(record, Buffer.concat(chunks), {
        signature: request.headers['x-webhook-signature'],
        timestamp: request.headers['x-webhook-timestamp'],
        delivery: request.headers['x-webhook-delivery'],
      })
      reply(result.duplicate ? 200 : 202, result)
    } catch (error) {
      // 只暴露协议校验错误；磁盘或内部异常不把本机路径等细节回传。
      if (!response.headersSent && !response.destroyed) reply(error.statusCode ?? 500, { error: error.statusCode ? error.message : 'Webhook内部错误' })
    }
  })
  server.requestTimeout = 10_000
  server.headersTimeout = 5_000
  // 这一行才真正监听TCP端口；等待成功后，调用方才能打印地址或发请求。
  await new Promise((resolve, reject) => {
    server.once('error', reject)
    server.listen(port, '127.0.0.1', () => { server.removeListener('error', reject); resolve() })
  })
  let closing
  return {
    url: 'http://127.0.0.1:' + server.address().port + '/webhooks',
    // 停服务时关闭连接；Host的Agent和日志由拥有它的调用方另行关闭。
    close: () => closing ??= new Promise((resolve, reject) => {
      server.close(error => error ? reject(error) : resolve())
      server.closeAllConnections()
    }),
  }
}
````

### 创建 [`src/webhook.mjs`](../../../harness-own/src/webhook.mjs)

````javascript
import { createHmac, createHash, timingSafeEqual } from 'node:crypto'

/** 自定义签名同时绑定时间、投递身份和原始正文；与GitHub协议不同。 */
export function webhookSignature(secret, body, { timestamp, delivery }) {
  return 'sha256=' + createHmac('sha256', secret).update(String(timestamp) + '\n' + delivery + '\n').update(body).digest('hex')
}

/** 给接入层提供明确状态码；持久化等未知错误仍由HTTP层报告500。 */
function reject(statusCode, message) { throw Object.assign(Error(message), { statusCode }) }

/** 验签与入队保持同步，避免同一进程的并发请求穿过检查后重复追加。 */
export function webhook({ secret, maxAge = 300_000 }) {
  if (!secret) throw Error('Webhook必须配置secret')
  return (record, body, { signature, timestamp, delivery }) => {
    // 时间使用毫秒字符串，投递身份限制为ASCII；两者都包含在HMAC中。
    if (!/^\d{13}$/.test(String(timestamp)) || !/^[a-zA-Z0-9_-]{1,100}$/.test(delivery ?? '') || Math.abs(Date.now() - Number(timestamp)) > maxAge) reject(401, 'Webhook身份或时间失效')
    const expected = webhookSignature(secret, body, { timestamp, delivery })
    // 固定ASCII格式后再比较字节，避免等字符数却不等字节数造成异常。
    if (!/^sha256=[a-f0-9]{64}$/.test(signature ?? '') || !timingSafeEqual(Buffer.from(signature), Buffer.from(expected))) reject(401, 'Webhook签名错误')
    const identity = 'webhook_' + delivery, digest = createHash('sha256').update(body).digest('hex')
    const previous = record.session.read().find(event => event.type === 'inbox/add' && event.data.id === identity)
    if (previous) {
      if (previous.data.webhookDigest !== digest) reject(409, '相同投递身份的正文不同')
      return { duplicate: true, inputId: identity }
    }
    let value
    try { value = JSON.parse(body.toString()) } catch { reject(400, 'Webhook正文不是合法JSON') }
    if (!value || typeof value.prompt !== 'string' || !value.prompt.trim()) reject(400, 'Webhook缺少非空prompt')
    if (record.agent.closed) reject(503, 'Agent已关闭')
    // 同一条持久日志既记录接收身份又保存任务；摘要用于识别身份复用但正文不同的请求。
    record.session.append('inbox/add', { id: identity, content: value.prompt, destination: 'turn', delivery, webhookDigest: digest })
    // 唤醒真实Agent领取任务，HTTP不等待模型执行结束；入队不等于执行成功。
    record.agent.pump()
    return { duplicate: false, inputId: identity }
  }
}
````

````javascript
/** 为原始字节生成自定义协议请求头，允许单项测试覆盖时间或投递身份。 */
function headers(body, delivery = 'delivery-1', timestamp = String(Date.now())) {
  return { 'content-type': 'application/json', 'x-webhook-delivery': delivery, 'x-webhook-timestamp': timestamp,
    'x-webhook-signature': webhookSignature(secret, body, { timestamp, delivery }) }
}

/** 每项HTTP测试使用真实临时Journal、Host和Agent，并在结束时关闭端口和删除目录。 */
async function fixture(run, options = {}) {
  const root = await mkdtemp(join(tmpdir(), 'webhook-061-')), host = new Host(root)
  let service
  try {
    const record = host.create({ cwd: root, owner: 'test' })
    service = await startWebhook(record, { secret, ...options })
    await run({ record, host, service, root })
  } finally {
    try { if (service) await service.close() }
    finally { try { await host.close() } finally { await rm(root, { recursive: true, force: true }) } }
  }
}

test('接收函数验证签名、时间和投递身份，重复请求只唤醒一次', () => {
  let pumps = 0
  const record = { session: new Events(), agent: { pump: () => pumps++ } }, body = Buffer.from('{"prompt":"review"}')
  const timestamp = String(Date.now()), delivery = 'delivery-1'
  const metadata = { timestamp, delivery, signature: webhookSignature(secret, body, { timestamp, delivery }) }
  assert.equal(webhook({ secret })(record, body, metadata).duplicate, false)
  assert.equal(webhook({ secret })(record, body, metadata).duplicate, true)
  assert.equal(pumps, 1)
  assert.throws(() => webhook({ secret })(record, Buffer.from('changed'), metadata), /签名/)
  assert.throws(() => webhook({ secret })(record, body, { ...metadata, delivery: 'delivery-2' }), /签名/)
  assert.throws(() => webhook({ secret })(record, body, { ...metadata, timestamp: String(Number(timestamp) - 1) }), /签名/)
})

test('真实HTTP投递触发Agent与离线模型，并在重开Host后保持去重', async () => {
  await fixture(async ({ record, host, service, root }) => {
    const body = Buffer.from('{"prompt":"来自HTTP的任务"}')
    const response = await fetch(service.url, { method: 'POST', headers: headers(body), body })
    assert.equal(response.status, 202)
    assert.deepEqual(await response.json(), { duplicate: false, inputId: 'webhook_delivery-1' })
    await record.agent.idle()
    assert.equal(record.session.read().findLast(event => event.type === 'message/assistant').data.content, '[离线演示] 来自HTTP的任务')
    assert.equal(record.session.read().findLast(event => event.type === 'turn/end').data.reason, 'completed')
    // 关闭并创建新的Host，从磁盘Journal恢复；不是只重建接收函数。
    await service.close()
    await host.close()
    const reopenedHost = new Host(root)
    let reopenedService
    try {
      const restored = reopenedHost.open(record.header.id, 'test')
      reopenedService = await startWebhook(restored, { secret })
      const duplicate = await fetch(reopenedService.url, { method: 'POST', headers: headers(body), body })
      assert.equal(duplicate.status, 200)
      assert.equal((await duplicate.json()).duplicate, true)
      await restored.agent.idle()
      assert.equal(restored.session.read().filter(event => event.type === 'inbox/add').length, 1)
      assert.equal(restored.session.read().filter(event => event.type === 'turn/start').length, 1)
    } finally {
      try { if (reopenedService) await reopenedService.close() } finally { await reopenedHost.close() }
    }
  })
})

test('同一进程并发重复投递只启动一回合，相同身份不同正文拒绝', async () => {
  await fixture(async ({ record, service }) => {
    const body = Buffer.from('{"prompt":"并发任务"}'), request = { method: 'POST', headers: headers(body), body }
    const replies = await Promise.all([fetch(service.url, request), fetch(service.url, request)])
    assert.deepEqual(replies.map(response => response.status).sort(), [200, 202])
    await record.agent.idle()
    assert.equal(record.session.read().filter(event => event.type === 'turn/start').length, 1)

    assert.equal(conflict.status, 409)
    assert.equal(record.session.read().filter(event => event.type === 'inbox/add').length, 1)
  })
})

test('错误签名、过期、无效JSON与空prompt都不入队', async () => {
  await fixture(async ({ record, service }) => {
    const body = Buffer.from('{"prompt":"review"}')
    const cases = [
      [body, { ...headers(body), 'x-webhook-signature': 'sha256=' + '0'.repeat(64) }, 401],
      [body, headers(body, 'expired', String(Date.now() - 600_000)), 401],
      [Buffer.from('{'), headers(Buffer.from('{')), 400],
      [Buffer.from('null'), headers(Buffer.from('null')), 400],
      [Buffer.from('{"prompt":"  "}'), headers(Buffer.from('{"prompt":"  "}')), 400],

  })
})

test('路由、方法、内容类型和正文大小受限，关闭后释放端口', async () => {

})

test('Agent关闭后拒绝新投递，持久化失败返回500且不报告接收成功', async () => {
  await fixture(async ({ record, service }) => {
    const body = Buffer.from('{"prompt":"review"}')
````

### 先看文件分工

### 步骤1：先在终端A启动监听

先按本节创建所有文件。在自己的PROJECT执行：

```sh
HARNESS_WEBHOOK_SECRET=local-demo-only bun src/webhook-server.mjs
```

终端会打印Webhook地址、独立数据目录、会话ID和模型说明。地址示例为http://127.0.0.1:随机端口/webhooks，必须复制本次真实输出，不能原样输入“随机端口”。真正监听的位置是webhook-http.mjs的server.listen(port, '127.0.0.1', ...)；端口0表示让系统分配空闲端口。设置HARNESS_WEBHOOK_PORT可选择固定端口。

服务默认持续等待请求，终端不会自动结束；观察完成后在终端A按Ctrl+C退出。程序关闭HTTP连接、Agent、Journal和锁，保留数据目录供检查。启动失败也会清理已创建的Host。它只监听本机，不直接接收互联网平台的投递，本章不配置公网入口。

### 步骤2：终端B发送真实HTTP请求

在同一个PROJECT另开终端，将示例端口12345替换成终端A打印的端口：

```sh
HARNESS_WEBHOOK_SECRET=local-demo-only bun scripts/send-webhook.mjs \
  http://127.0.0.1:12345/webhooks '帮我处理这条通知' delivery-1
```

预期终端B收到HTTP 202和duplicate:false、inputId:webhook_delivery-1。这表示已持久入队，不表示模型已处理完成。终端A稍后打印“Agent回复：[离线演示] 帮我处理这条通知”以及“Agent回合结束：completed”。响应与日志在两个终端的显示先后可能不同。

这里不是计数替身：HTTP、Host、Journal、Inbox、Agent和runTurn都是真实实现，只有外部模型被明确替换为OfflineModel，所以无需DeepSeek API Key。换成真实模型需要通过Host的modelFactory接入；不要把离线回复误当成远程模型输出。

### 步骤3：验证重复投递与冲突

再次运行完全相同的发送命令：预期HTTP 200、duplicate:true，不再追加任务或启动新回合。保持delivery-1不变，把任务正文改成另一句话：预期HTTP 409，提示相同投递身份的正文不同。换成delivery-2则是一个新任务。

### 完整时序：哪里监听，哪里调用模型？


### 自定义HTTP协议与错误响应

| 项目 | 本章约定 |
| --- | --- |
| 路由与方法 | POST /webhooks |
| 内容类型与正文 | application/json，非空字符串prompt，默认最多64 KiB |
| x-webhook-timestamp | 当前毫秒时间戳，默认前后5分钟内 |
| x-webhook-delivery | 1～100个ASCII字母、数字、下划线或短横线 |
| x-webhook-signature | sha256=后接小写十六进制HMAC |
| 签名输入 | timestamp + 换行 + delivery + 换行 + 原始body字节 |
| 首次接收 / 重复接收 | 202 / 200 |
| 校验与接入错误 | 400正文无效、401验签或时间失败、404路由、405方法、409身份冲突、413正文过大、415类型不符、503 Agent关闭 |
| 内部错误 | 500；例如Journal写入失败，不回传内部路径 |

这个格式不是GitHub官方Webhook协议，外部发送方必须按本章格式签名。时间、delivery和原始正文都在HMAC范围内，不能签名后改JSON空白、换投递ID或更新时间；重试时由持有secret的发送方重新签名。

### 从磁盘恢复与测试证据

6项测试覆盖：处理函数验签与元数据防篡改；真实HTTP触发真实Agent及离线回复；关闭并重开Host后读取Journal去重；同进程并发重复投递；同ID不同正文冲突；错误正文、请求类型、大小及内部失败不误报成功。它们没有连接外部模型或公网账号系统。

要手工重开同一会话，在终端A正常退出后，使用原输出的数据目录和会话ID：

```sh
HARNESS_WEBHOOK_SECRET=local-demo-only \
HARNESS_WEBHOOK_HOME='/替换为原数据目录' \
HARNESS_WEBHOOK_SESSION='替换为原session_ID' \
bun src/webhook-server.mjs
```

服务会打印新的端口；终端B使用新地址重发原任务与原delivery，仍应返回duplicate:true。集成测试验证的是关闭Host后创建新Host并重新读盘，未做进程崩溃故障注入。已入队后崩溃、模型执行失败和外部工具副作用，不由HTTP接收去重保证“恰好执行一次”；本章也没有自动重放未完成任务、死信或多进程共享同一会话的机制。

### 调试顺序

1. 用“061：本节测试”看自动断言；用“061：Webhook监听服务”看手工链路，配置内使用local-demo-only演示secret。
2. 在webhook-http.mjs的server.listen设断点，确认先启动监听，再去另一个终端发请求。
3. 在receive(record, ...)前观察原始body和请求头；在webhook.mjs的previous查询、append与pump停下，确认只有首次投递入队。
4. 在agent.mjs的inbox.claim和runTurn停下，进入loop.mjs的model.stream，确认是Agent消费任务，而不是模型调用Webhook。
5. 在终端A观察message/assistant与turn/end；HTTP 202不代替这一步。默认请求读取超时10秒、签名时间窗5分钟，长时间停断点后需重新发送签名请求。
6. Ctrl+C或停止调试，确认端口释放。日志仍留在打印的数据目录，可供下次重开检查。
